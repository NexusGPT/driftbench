import assert from 'node:assert/strict';import {selectLabellerThreshold as select} from './candidate';
const set={name:'x',labels:[{name:'a',description:''},{name:'b',description:''}]};const obs=[{gold:'a',answer:'a',confidence:.8},{gold:'a',answer:'b',confidence:.2}];
assert.equal(select(obs,set,{thresholds:[.9,.7,.5,0],minAccuracy:.9})!.threshold,.5);
assert.equal(select(obs,set,{thresholds:[0,.5],minAccuracy:.9,minCoverage:.6}),null);
assert.equal(select(obs,set,{thresholds:[.9],minAccuracy:0}),null);
assert.throws(()=>select(obs,set,{thresholds:[0],minAccuracy:NaN}));assert.throws(()=>select(obs,set,{thresholds:[0],minAccuracy:0,minCoverage:2}));console.log('threshold selector regression passed');
