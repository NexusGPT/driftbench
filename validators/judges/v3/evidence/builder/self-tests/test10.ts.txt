import assert from 'node:assert/strict';import {labellerNominalAlpha as alpha} from './candidate';
const set={name:'x',labels:[{name:'a',description:''},{name:'b',description:''}]};
const r=alpha([['a','b','a'],['a','b',null],['b',null,null]],set);
assert.ok(Math.abs(r.alpha!-1/3)<1e-12);assert.equal(r.observed,.4);assert.equal(r.expected,.6);
assert.equal(r.pairable,5);assert.equal(r.units,3);assert.equal(r.raters,3);assert.equal(r.missing,3);assert.equal(r.unpairableUnits,1);
assert.equal(alpha([['a'],['a']],set).alpha,1);assert.equal(alpha([['a'],[null]],set).status,'insufficient');
assert.throws(()=>alpha([['a'],['bad']],set));assert.throws(()=>alpha([['a'],[undefined as any]],set));assert.throws(()=>alpha([['a'],[]],set));assert.throws(()=>alpha([['a']],set));console.log('nominal alpha regression passed');
