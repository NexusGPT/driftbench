import { createHash } from 'node:crypto';
/** One nominal class. Names are compared exactly; descriptions participate in replay identity. */
export interface LabelDefinition {
  readonly name: string;
  readonly description: string;
}

export interface LabelSet {
  readonly name: string;
  readonly labels: readonly LabelDefinition[];
}

export interface LabellerRequest {
  readonly text: string;
  readonly labelSet: LabelSet;
  readonly context?: string;
}

/** `none` is a valid class; only null abstains. Other strings can be recorded but score invalid. */
export type LabellerAnswer = string | null;

export interface Labeller {
  readonly id: string;
  label(request: LabellerRequest): Promise<LabellerAnswer> | LabellerAnswer;
}

export interface LabellerScore {
  readonly total: number;
  readonly correct: number;
  readonly agreement: number;
  readonly kappa: number;
  readonly abstentions: number;
  readonly invalid: number;
  /** Valid class answers / all observations, including the implicit `none` class. */
  readonly coverage: number;
}

export interface LabellerReplayEntry {
  readonly request: LabellerRequest;
  readonly labellerId: string;
  readonly answer: LabellerAnswer;
}

export function labelSetHash(set: LabelSet): string {
  domain(set);
  return createHash('sha256').update(JSON.stringify([set.name,set.labels.map(l=>[l.name,l.description])])).digest('hex');
}
export function labellerRequestKey(request: LabellerRequest, labellerId: string): string {
  return createHash('sha256').update(JSON.stringify([labelSetHash(request.labelSet),request.text,request.context===undefined?[]:[request.context],labellerId])).digest('hex');
}
function domain(set: LabelSet): string[] {
  if (!set || typeof set.name !== 'string' || !set.name.trim() || !Array.isArray(set.labels) || !set.labels.length) throw new Error('Invalid label set');
  const names: string[]=[];
  for (const label of set.labels) {
    if (!label || typeof label.name !== 'string' || !label.name.trim() || typeof label.description !== 'string' || names.includes(label.name)) throw new Error('Invalid label definition');
    names.push(label.name);
  }
  if (!names.includes('none')) names.push('none');
  return names;
}
function nominal(gold: readonly string[], answers: readonly LabellerAnswer[], set: LabelSet) {
  const labels=domain(set);
  if (!gold.length || gold.length!==answers.length) throw new Error('Empty or unequal observations');
  const rows=Array(labels.length+2).fill(0), cols=Array(labels.length+2).fill(0);
  const pairs: [number,number][]=[];
  let correct=0, abstentions=0, invalid=0;
  for (let i=0;i<gold.length;i++) {
    const a=labels.indexOf(gold[i]); if(a<0) throw new Error('Unknown gold');
    const answer=answers[i];
    let b=typeof answer==='string'?labels.indexOf(answer):-1;
    if(answer===null) { b=labels.length; abstentions++; }
    else if(b<0) { b=labels.length+1; invalid++; }
    rows[a]++; cols[b]++; pairs.push([a,b]); if(a===b) correct++;
  }
  const n=gold.length, observed=correct/n;
  const expected=rows.reduce((sum,r,i)=>sum+r*cols[i],0)/(n*n);
  const kappa=expected===1 ? 1 : (observed-expected)/(1-expected);
  return {labels,rows,cols,pairs,observed,expected,score:{total:n,correct,agreement:observed,kappa,abstentions,invalid,coverage:(n-abstentions-invalid)/n}};
}
export function scoreLabeller(gold: readonly string[], answers: readonly LabellerAnswer[], labelSet: LabelSet): LabellerScore {
  return nominal(gold,answers,labelSet).score;
}
export function createLabellerReplay(entries: readonly LabellerReplayEntry[], id: string): Labeller {
  const snapshot=new Map<string,LabellerAnswer>();
  for(const entry of entries) {
    const key=labellerRequestKey(entry.request,entry.labellerId);
    if(snapshot.has(key) && snapshot.get(key)!==entry.answer) throw new Error('Conflicting replay answers');
    snapshot.set(key,entry.answer);
  }
  return {id,label(request) {
    const key=labellerRequestKey(request,id);
    if(!snapshot.has(key)) throw new Error('Replay miss');
    return snapshot.get(key)!;
  }};
}

/** Advanced labeller agreement analysis. Statistical methods and limitations are documented per API. */
interface Interval { estimate: number; lower: number; upper: number }
interface KappaResult { kappa: number; observed: number; expected: number; n: number }

export interface LabellerKappaInterval {
  estimate: number; lower: number | null; upper: number | null;
  standardError: number | null; varianceFactor: number | null;
  confidence: number; total: number; status: 'estimated' | 'degenerate';
  method: 'multinomial-delta';
}
function critical(confidence:number):number {
  if(!Number.isFinite(confidence)||confidence<=0||confidence>=1) throw new Error('Confidence must be strictly between zero and one');
  if(confidence<1e-8) return confidence*Math.sqrt(Math.PI/2);
  const p=(1-confidence)/2;
  if(p<.02425) {
    const q=Math.sqrt(-2*Math.log(p));
    return -(((((-.007784894002430293*q-.3223964580411365)*q-2.400758277161838)*q-2.549732539343734)*q+4.374664141464968)*q+2.938163982698783)/((((.007784695709041462*q+.3224671290700398)*q+2.445134137142996)*q+3.754408661907416)*q+1);
  }
  const q=p-.5,r=q*q;
  return -(((((-39.69683028665376*r+220.9460984245205)*r-275.9285104469687)*r+138.357751867269)*r-30.66479806614716)*r+2.506628277459239)*q/(((((-54.47609879822406*r+161.5858368580409)*r-155.6989798598866)*r+66.80131188771972)*r-13.28068155288572)*r+1);
}
export function labellerKappaInterval(gold: readonly string[], answers: readonly LabellerAnswer[], labelSet: LabelSet, confidence?: number): LabellerKappaInterval {
  const c=confidence??.95,z=critical(c),data=nominal(gold,answers,labelSet),n=gold.length;
  const {observed:po,expected:pe,rows,cols,pairs,score}=data;
  const base={estimate:score.kappa,confidence:c,total:n,method:'multinomial-delta' as const};
  const degenerate={...base,lower:null,upper:null,standardError:null,varianceFactor:null,status:'degenerate' as const};
  if(n<2||pe===1) return degenerate;
  const gradients=pairs.map(([a,b])=>(a===b?1:0)/(1-pe)+(po-1)*(cols[a]/n+rows[b]/n)/(1-pe)**2);
  const mean=gradients.reduce((a,b)=>a+b,0)/n;
  const varianceFactor=gradients.reduce((a,b)=>a+(b-mean)**2,0)/n;
  if(varianceFactor<=1e-15) return degenerate;
  const standardError=Math.sqrt(varianceFactor/n);
  return {...base,lower:Math.max(-1,score.kappa-z*standardError),upper:Math.min(1,score.kappa+z*standardError),standardError,varianceFactor,status:'estimated'};
}
export function binaryLabellerPabak(gold: readonly string[], answers: readonly LabellerAnswer[], categories: readonly [string, string]): { total: number; agreement: number; pabak: number } {
  if(categories.length!==2 || categories.some(x=>typeof x!=='string'||!x.trim()) || categories[0]===categories[1]) throw new Error('Two distinct binary categories required');
  if(!gold.length || gold.length!==answers.length) throw new Error('Empty or unequal observations');
  let correct=0;
  for(let i=0;i<gold.length;i++) {
    if(!categories.includes(gold[i]) || !categories.includes(answers[i] as string)) throw new Error('Observation outside binary domain');
    if(gold[i]===answers[i]) correct++;
  }
  const agreement=correct/gold.length;
  return {total:gold.length,agreement,pabak:2*agreement-1};
}
export function weightedLabellerKappa(gold: readonly string[], answers: readonly LabellerAnswer[], orderedLabels: readonly string[], weights?: 'linear' | 'quadratic'): KappaResult {
  const mode=weights??'linear',k=orderedLabels.length,n=gold.length;
  if(k<2 || new Set(orderedLabels).size!==k || orderedLabels.some(x=>typeof x!=='string'||!x.trim()) || !['linear','quadratic'].includes(mode)) throw new Error('Invalid ordinal domain or weight');
  if(!n||n!==answers.length) throw new Error('Empty or unequal observations');
  const rows=Array(k).fill(0),cols=Array(k).fill(0);
  const distance=(a:number,b:number)=>mode==='linear'?Math.abs(a-b)/(k-1):((a-b)/(k-1))**2;
  let observedDisagreement=0;
  for(let i=0;i<n;i++) {
    const a=orderedLabels.indexOf(gold[i]),b=orderedLabels.indexOf(answers[i] as string);
    if(a<0||b<0) throw new Error('Observation outside ordinal domain');
    rows[a]++;cols[b]++;observedDisagreement+=distance(a,b)/n;
  }
  let expectedDisagreement=0;
  for(let a=0;a<k;a++) for(let b=0;b<k;b++) expectedDisagreement+=rows[a]*cols[b]*distance(a,b)/(n*n);
  return {n,observed:1-observedDisagreement,expected:1-expectedDisagreement,kappa:expectedDisagreement===0?1:1-observedDisagreement/expectedDisagreement};
}

export type LabellerPredictionColumn = { kind: 'label'; label: string } | { kind: 'abstain' } | { kind: 'invalid' };
export interface LabellerClassMetrics {
  label: string; support: number; predicted: number; tp: number; fp: number; fn: number;
  precision: number | null; recall: number | null; f1: number | null;
}
export interface LabellerConfusion {
  score: LabellerScore; labels: readonly string[];
  predictionColumns: readonly LabellerPredictionColumn[];
  confusion: readonly (readonly number[])[]; byLabel: readonly LabellerClassMetrics[];
}
export function labellerConfusion(gold: readonly string[], answers: readonly LabellerAnswer[], labelSet: LabelSet): LabellerConfusion {
  const {labels,pairs,score}=nominal(gold,answers,labelSet);
  const confusion=labels.map(()=>Array(labels.length+2).fill(0));
  for(const [a,b] of pairs) confusion[a][b]++;
  const predictionColumns:LabellerPredictionColumn[]=[...labels.map(label=>({kind:'label' as const,label})),{kind:'abstain'},{kind:'invalid'}];
  const byLabel=labels.map((label,i)=>{
    const support=confusion[i].reduce((a,b)=>a+b,0), predicted=confusion.reduce((a,row)=>a+row[i],0),tp=confusion[i][i];
    const fp=predicted-tp,fn=support-tp;
    return {label,support,predicted,tp,fp,fn,precision:predicted?tp/predicted:null,recall:support?tp/support:null,f1:2*tp+fp+fn?2*tp/(2*tp+fp+fn):null};
  });
  return {score,labels,predictionColumns,confusion,byLabel};
}

export interface LabellerLabelMapping { source: string; target: string }
export interface LabellerMappingOptions {
  goldSet: LabelSet; answerSet: LabelSet; targetSet: LabelSet;
  goldMap: readonly LabellerLabelMapping[]; answerMap: readonly LabellerLabelMapping[];
}
export function scoreMappedLabeller(gold: readonly string[], answers: readonly LabellerAnswer[], options: LabellerMappingOptions): LabellerScore {
  const targets=domain(options.targetSet);
  function mapping(set:LabelSet,entries:readonly LabellerLabelMapping[]) {
    const sources=domain(set),map=new Map<string,string>();
    for(const entry of entries) {
      if(!sources.includes(entry.source)||!targets.includes(entry.target)||map.has(entry.source)) throw new Error('Invalid or duplicate mapping');
      map.set(entry.source,entry.target);
    }
    if(map.size!==sources.length) throw new Error('Mapping must cover every source including none');
    return map;
  }
  const gm=mapping(options.goldSet,options.goldMap),am=mapping(options.answerSet,options.answerMap);
  const mappedGold=gold.map(g=>{if(!gm.has(g)) throw new Error('Unknown source gold');return gm.get(g)!;});
  const mappedAnswers=answers.map(a=>a===null?null:am.has(a)?am.get(a)!:undefined as unknown as LabellerAnswer);
  return scoreLabeller(mappedGold,mappedAnswers,options.targetSet);
}
