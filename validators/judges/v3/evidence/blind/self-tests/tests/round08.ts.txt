import assert from 'node:assert/strict';
import {labellerCoverageCurve as curve} from '../candidate';
const set={name:'s',labels:[{name:'a',description:''},{name:'b',description:''}]};
const obs=[{gold:'a',answer:'a',confidence:.8},{gold:'b',answer:'a',confidence:.8},{gold:'b',answer:'b',confidence:.4},{gold:'a',answer:null,confidence:1},{gold:'a',answer:'bad',confidence:1}];
const [low,high,empty]=curve(obs,set,[1,.8,.8,0]);assert.equal(low.accepted,3);assert.equal(low.belowThreshold,0);assert.equal(high.accepted,2);assert.equal(high.acceptedAccuracy,.5);assert.equal(high.coverage,.4);assert.equal(high.abstentions,1);assert.equal(high.invalid,1);assert.equal(high.belowThreshold,1);
assert.ok(Math.abs(high.acceptedAccuracyInterval!.lower-(1-Math.sqrt(.975)))<1e-12);assert.ok(Math.abs(high.acceptedAccuracyInterval!.upper-Math.sqrt(.975))<1e-12);
assert.equal(empty.acceptedAccuracy,null);assert.equal(empty.acceptedAccuracyInterval,null);
assert.throws(()=>curve([{...obs[0],confidence:NaN}],set,[0]));assert.throws(()=>curve(obs,set,[]));assert.throws(()=>curve(obs,set,[-.1]));
const extreme=curve([obs[0]],set,[0],1-Number.EPSILON)[0];assert.ok(extreme.acceptedAccuracyInterval!.lower>0);
console.log('round08 coverage and exact binomial intervals passed');
