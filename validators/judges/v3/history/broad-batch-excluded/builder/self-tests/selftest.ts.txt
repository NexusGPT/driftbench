import assert from 'node:assert/strict';
import * as c from './candidate';
const set={name:'test',labels:[{name:'a',description:'A'},{name:'b',description:'B'}]};
const near=(a:number,b:number)=>assert.ok(Math.abs(a-b)<1e-8,`${a} differs from ${b}`);
const tests: Array<[string,()=>unknown]> = [
 ['nominal missing categories retain denominator',()=>{const s=c.scoreLabeller(['a','b','none','a'],['a',null,'none','oops'],set);assert.deepEqual(s,{total:4,correct:2,agreement:.5,kappa:5/13,abstentions:1,invalid:1,coverage:.5});}],
 ['negative agreement and constant convention',()=>{near(c.scoreLabeller(['a','b'],['b','a'],set).kappa,-1);near(c.scoreLabeller(['a'],['a'],set).kappa,1);assert.throws(()=>c.scoreLabeller([],[],set));}],
 ['hash identity distinguishes definitions and optional context',()=>{assert.match(c.labelSetHash(set),/^[0-9a-f]{64}$/);assert.notEqual(c.labelSetHash(set),c.labelSetHash({...set,labels:[...set.labels].reverse()}));assert.notEqual(c.labellerRequestKey({text:'x',labelSet:set},'l'),c.labellerRequestKey({text:'x',labelSet:set,context:''},'l'));}],
 ['replay snapshot and miss',()=>{const request={text:'x',labelSet:structuredClone(set)};const entries=[{request,labellerId:'l',answer:'a'}];const replay=c.createLabellerReplay(entries,'l');request.text='y';assert.equal(replay.label({text:'x',labelSet:set}),'a');assert.throws(()=>replay.label({text:'z',labelSet:set}));assert.throws(()=>c.createLabellerReplay([{request,labellerId:'l',answer:'a'},{request,labellerId:'l',answer:'b'}],'l'));}],
 ['binary pabak domain',()=>{assert.equal(c.binaryLabellerPabak(['a','b'],['a','a'],['a','b']).pabak,0);assert.throws(()=>c.binaryLabellerPabak(['a'],[null],['a','b']));}],
 ['weighted kappa normalization',()=>{const r=c.weightedLabellerKappa(['a','b','c'],['a','c','b'],['a','b','c']);near(r.observed,2/3);near(r.expected,5/9);near(r.kappa,.25);assert.throws(()=>c.weightedLabellerKappa(['a'],['a'],['a','b'],'bad' as any));}],
 ['alpha missing and pairability',()=>{const a=c.labellerNominalAlpha([['a',null,'a'],['b',null,null]],set);assert.equal(a.pairable,2);assert.equal(a.missing,3);assert.equal(a.unpairableUnits,2);near(a.alpha!,0);assert.equal(c.labellerNominalAlpha([[null],[null]],set).status,'insufficient');}],
 ['confusion typed reserved categories',()=>{const x=c.labellerConfusion(['a','b'],[null,'bad'],set);assert.equal(x.confusion[0][3],1);assert.equal(x.confusion[1][4],1);assert.equal(x.byLabel[0].precision,null);assert.equal(x.byLabel[0].recall,0);}],
 ['coverage exact endpoint and threshold tie',()=>{const obs=[{gold:'a',answer:'a',confidence:.8},{gold:'b',answer:'b',confidence:.9}];const p=c.labellerCoverageCurve(obs,set,[1,.5,.5,0]);assert.equal(p.length,3);near(p[0].acceptedAccuracyInterval!.lower,Math.sqrt(.025));assert.equal(p[2].acceptedAccuracy,null);assert.equal(c.selectLabellerThreshold(obs,set,{thresholds:[.5,0],minAccuracy:1})!.threshold,0);}],
 ['explicit mapping validates total domains',()=>{const opts={goldSet:set,answerSet:set,targetSet:set,goldMap:['a','b','none'].map(source=>({source,target:source})),answerMap:['a','b','none'].map(source=>({source,target:source}))};near(c.scoreMappedLabeller(['a','none'],['a',null],opts).agreement,.5);assert.throws(()=>c.scoreMappedLabeller(['a'],['a'],{...opts,goldMap:opts.goldMap.slice(0,2)}));}],
 ['delta interval planner consistency',()=>{const gold=['a','a','a','a','b','b','b','b'];const ans=['a','a','a','b','a','b','b','b'];const ci=c.labellerKappaInterval(gold,ans,set);near(ci.estimate,.5);near(ci.varianceFactor!,.75);const p=c.planLabellerKappaSample([[.375,.125],[.125,.375]],{margin:.1});assert.equal(p.sampleSize,289);assert.equal(c.labellerKappaInterval(['a'],['a'],set).status,'degenerate');}],
 ['exact fixed margin permutation',()=>{const r=c.exactLabellerKappaTest(['a','a','b','b'],['a','a','b','b'],set);assert.equal(r.permutations,6);assert.equal(r.extreme,2);near(r.pValue,1/3);assert.equal(c.exactLabellerKappaTest(['a','a'],[null,null],set).pValue,1);}],
];
let failures=0;
for(const [name,fn] of tests){try{fn();console.log('PASS',name);}catch(e){failures++;console.log('FAIL',name,String(e));}}
console.log(JSON.stringify({tests:tests.length,failures}));
process.exitCode=failures?1:0;
